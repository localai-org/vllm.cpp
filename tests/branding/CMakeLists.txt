cmake_minimum_required(VERSION 3.24)
project(branding_consumer LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(ENGINE_SOURCE)
  add_subdirectory("${ENGINE_SOURCE}" engine)
else()
  find_package(LumenEngine CONFIG REQUIRED)
endif()
foreach(alias LumenEngine::LumenEngine LumenEngine::shared)
  if(TARGET ${alias})
    get_target_property(kind ${alias} TYPE)
    if(NOT kind STREQUAL "SHARED_LIBRARY")
      message(FATAL_ERROR "${alias} must select the shared C ABI library")
    endif()
  endif()
endforeach()
add_executable(branding_old legacy_consumer.cpp)
foreach(name new shared)
  add_executable(branding_${name} consumer.cpp)
endforeach()
target_link_libraries(branding_old PRIVATE vllm::shared)
target_link_libraries(branding_new PRIVATE LumenEngine::LumenEngine)
target_link_libraries(branding_shared PRIVATE LumenEngine::shared)
add_executable(branding_old_c legacy_consumer.c)
target_link_libraries(branding_old_c PRIVATE vllm::shared)
add_executable(branding_c consumer.c)
target_link_libraries(branding_c PRIVATE LumenEngine::shared)
if(ENGINE_SOURCE)
  add_executable(branding_static legacy_consumer.cpp)
  target_link_libraries(branding_static PRIVATE vllm::vllm)
else()
  # A second lookup must preserve the existing imported targets.
  find_package(LumenEngine CONFIG REQUIRED)
endif()
enable_testing()
foreach(name old old_c new shared c)
  add_test(NAME branding_${name} COMMAND branding_${name})
endforeach()
if(TARGET branding_static)
  add_test(NAME branding_static COMMAND branding_static)
endif()
# Run the existing ABI smoke through the same built or imported shared target.
add_executable(branding_dlopen ../capi/test_dlopen.cpp ../doctest_main.cpp)
target_include_directories(branding_dlopen PRIVATE
  "${CMAKE_CURRENT_LIST_DIR}/../../third_party"
  "$<TARGET_PROPERTY:vllm::shared,INTERFACE_INCLUDE_DIRECTORIES>")
target_compile_definitions(branding_dlopen PRIVATE
  VLLM_SHARED_LIB_PATH="$<TARGET_FILE:vllm::shared>")
target_link_libraries(branding_dlopen PRIVATE ${CMAKE_DL_LIBS})
add_test(NAME branding_dlopen COMMAND branding_dlopen)
if(UNIX AND NOT APPLE)
  add_test(NAME branding_exports
    COMMAND ${CMAKE_COMMAND} "-DVLLM_SHARED_LIB=$<TARGET_FILE:vllm::shared>"
      -P "${CMAKE_CURRENT_LIST_DIR}/../../cmake/VerifyExports.cmake")
endif()
